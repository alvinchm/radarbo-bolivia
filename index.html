Script Python con Selectores Reales (`scraper_bolivia.py`)

Extrae en vivo desde Shopify JSON (sologamerbolivia.com/products.json), iriscomputer.bo y cyrexstore.com

import json, re
from datetime import datetime
import requests
from bs4 import BeautifulSoup

# Tasa del día publicada en Cyrex Store (15/10/2026)
TASA_USD_BS = 11.80
UMBRAL_OFERTON_CRITICO = 45.0

HEADERS = {"User-Agent": "Mozilla/5.0 (Windows NT 10.0; Win64; x64) Chrome/129.0.0.0"}

def scrape_solo_gamer_shopify():
    """Solo Gamer Bolivia usa Shopify: expone su catálogo real en /products.json"""
    url = "https://sologamerbolivia.com/products.json?limit=250"
    r = requests.get(url, headers=HEADERS, timeout=15)
    productos = []
    for p in r.json().get("products", []):
        v = p["variants"][0]
        precio_actual = float(v["price"])
        precio_antes = float(v["compare_at_price"]) if v.get("compare_at_price") else precio_actual
        desc = round(((precio_antes - precio_actual) / precio_antes) * 100, 1) if precio_antes > precio_actual else 0.0
        productos.append({
            "tienda": "Solo Gamer BO",
            "nombre": p["title"],
            "precio_bs": precio_actual,
            "precio_referencia_bs": precio_antes,
            "descuento_pct": desc,
            "es_oferton_45": desc >= UMBRAL_OFERTON_CRITICO,
            "url": f"https://sologamerbolivia.com/products/{p['handle']}"
        })
    return productos

def scrape_iris_computer(categoria_id=9):
    """Extrae productos en USD de iriscomputer.bo y los convierte a Bs."""
    url = f"https://www.iriscomputer.bo/productos.php?categoria={categoria_id}"
    soup = BeautifulSoup(requests.get(url, headers=HEADERS, timeout=15).text, "html.parser")
    items = []
    for texto in soup.stripped_strings:
        if texto.startswith("$") and len(texto) < 10:
            usd = float(texto.replace("$", "").replace(".", "").replace(",", ".").strip())
            items.append({"precio_usd": usd, "precio_bs": round(usd * TASA_USD_BS)})
    return items

if __name__ == "__main__":
    datos = scrape_solo_gamer_shopify()
    with open("precios_bolivia_reales.json", "w", encoding="utf-8") as f:
        json.dump(datos, f, ensure_ascii=False, indent=2)
    print(f"Extraídos {len(datos)} productos reales en vivo.")
